    /* Firestore treats '/' as a path separator, so doc(db,'overrides','node/123') resolves to a
     * three-segment path and THROWS. 6,595 location ids were OSM-derived and contained a slash —
     * Speedway entirely, plus every metro and public chain — so every override and tip edit made
     * from this panel silently failed for a quarter of the map. This panel had NO sanitising at
     * all. The data files now carry the safe form (original preserved in meta.srcId), but reads
     * still normalise so any override created earlier under a raw id keeps resolving. */
    function fsId(id){
      return String(id == null ? '' : id).replace(/\//g, '__');
    }
